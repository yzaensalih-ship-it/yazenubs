  <script>
    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.register('/uv/uv.sw.js', { scope: __uv$config.prefix });
    }

    document.getElementById('proxy-form').addEventListener('submit', (e) => {
      e.preventDefault();
      let url = document.getElementById('address').value.trim();
      if (!url.startsWith('http://') && !url.startsWith('https://')) {
        url = url.includes('.') ? 'https://' + url : 'https://www.google.com/search?q=' + encodeURIComponent(url);
      }
      window.location.href = __uv$config.prefix + __uv$config.encodeUrl(url);
    });
  </script>
